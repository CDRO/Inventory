+// The HTTP surface of docs/specs/17-expiry-notifications.md: read the
+// storage's digest configuration, save it, and prove it works.
+//
+// One rule shapes every response here: **the stored token is never returned**.
+// The read reports whether one is set and nothing more, which is why the
+// response type below has no token field at all rather than a hidden one —
+// a field tagged `json:"-"` is one careless edit away from being serialized,
+// and an absent field is not.
+//
+// The routes sit under /api/storages/{storage_id} like every other
+// storage-scoped route, so RequireStorageMember gates them and an unknown
+// storage and an inaccessible one are the same 404
+// (docs/specs/03-auth-and-multi-tenancy.md). Any member may configure
+// notifications: rights inside a storage are flat, and the digest is about
+// shared food.
+
+// defaultSendHour matches notification_settings.send_hour's column default.
+// It is repeated here for the one case that never reaches the column: the
+// read of a storage that has never been configured.
+const defaultSendHour = 8
+
+// NotificationStore is the slice of the store the notification handlers use.
+type NotificationStore interface {
+	NotificationSettingsFor(ctx context.Context, storageID uuid.UUID) (*store.NotificationSettings, error)
+	SaveNotificationSettings(ctx context.Context, storageID uuid.UUID, in store.NotificationSettingsInput) (*store.NotificationSettings, error)
+}
+
+// Notifier delivers a test message and records its outcome.
+// *notify.Service satisfies it.
+type Notifier interface {
+	SendTest(ctx context.Context, settings store.NotificationSettings) error
+}
+
+// NotificationHandler serves the notification-settings routes.
+type NotificationHandler struct {
+	store    NotificationStore
+	notifier Notifier
+	errors   *ErrorWriter
+}
+
+// NewNotificationHandler wires the handlers to a store, the delivery service
+// and the one error writer. A nil notifier leaves reads and writes working
+// and makes the test route absent — the same "absent collaborator, absent
+// route" rule the rest of the router follows.
+func NewNotificationHandler(s NotificationStore, notifier Notifier, errs *ErrorWriter) *NotificationHandler {
+	return &NotificationHandler{store: s, notifier: notifier, errors: errs}
+}
+
+// notificationSettingsResponse is what a caller sees.
+//
+// HasToken rather than Token, deliberately and permanently: a stored token is
+// never returned by any read (docs/specs/17-expiry-notifications.md). The UI
+// needs only to say "a token is saved" so somebody can decide whether to
+// replace it.
+//
+// LastResult is a curated summary written by internal/notify — never a
+// transport error's own text, which would carry the request URL and, for the
+// gotify kind, the token inside it.
+type notificationSettingsResponse struct {
+	Enabled     bool       `json:"enabled"`
+	Kind        string     `json:"kind"`
+	URL         string     `json:"url"`
+	HasToken    bool       `json:"has_token"`
+	SendHour    int        `json:"send_hour"`
+	IncludeSoon bool       `json:"include_soon"`
+	LastRunAt   *time.Time `json:"last_run_at"`
+	LastResult  *string    `json:"last_result"`
+}
+
+func newNotificationSettingsResponse(s store.NotificationSettings) notificationSettingsResponse {
+	return notificationSettingsResponse{
+		Enabled:     s.Enabled,
+		Kind:        s.Kind,
+		URL:         s.URL,
+		HasToken:    s.Token != "",
+		SendHour:    s.SendHour,
+		IncludeSoon: s.IncludeSoon,
+		LastRunAt:   s.LastRunAt,
+		LastResult:  s.LastResult,
+	}
+}
+
+// Get serves GET /api/storages/{storage_id}/notification-settings.
+//
+// A storage that has never been configured gets the documented defaults with
+// enabled false, not a 404: "you have not set this up" is the normal state
+// for most households, and the settings form needs something to render. The
+// 404s on this route come from the membership gate, where they belong.
+func (h *NotificationHandler) Get(w http.ResponseWriter, r *http.Request) {
+	storageID, ok := StorageIDFrom(r.Context())
+	if !ok {
+		h.errors.WriteError(w, r, Internal(errNoStorageInContext))
+		return
+	}
+
+	settings, err := h.store.NotificationSettingsFor(r.Context(), storageID)
+	if errors.Is(err, store.ErrNotFound) {
+		writeJSON(w, http.StatusOK, notificationSettingsResponse{
+			Kind:     store.NotificationNtfy,
+			SendHour: defaultSendHour,
+		})
+		return
+	}
+	if err != nil {
+		h.errors.WriteError(w, r, Internal(err))
+		return
+	}
+	writeJSON(w, http.StatusOK, newNotificationSettingsResponse(*settings))
+}
+
+// Put serves PUT /api/storages/{storage_id}/notification-settings.
+//
+// The token is the one three-state field: omitting it keeps whatever is
+// stored, sending null clears it, and sending a string replaces it. The
+// distinction has to exist because the value is never read back — a form that
+// re-saves the hour must not wipe a token it was never shown, and somebody
+// who wants the token gone still needs a way to say so. json.RawMessage is
+// how absent and null are told apart, the same idiom PatchBatchExpiry uses
+// for the same reason.
+func (h *NotificationHandler) Put(w http.ResponseWriter, r *http.Request) {
+	storageID, ok := StorageIDFrom(r.Context())
+	if !ok {
+		h.errors.WriteError(w, r, Internal(errNoStorageInContext))
+		return
+	}
+
+	var body struct {
+		Enabled     bool            `json:"enabled"`
+		Kind        string          `json:"kind"`
+		URL         string          `json:"url"`
+		Token       json.RawMessage `json:"token"`
+		SendHour    *int            `json:"send_hour"`
+		IncludeSoon bool            `json:"include_soon"`
+	}
+	if failure := decodeJSON(w, r, &body); failure != nil {
+		h.errors.WriteError(w, r, failure)
+		return
+	}
+
+	fields := map[string][]string{}
+	switch body.Kind {
+	case store.NotificationNtfy, store.NotificationGotify, store.NotificationWebhook:
+	default:
+		fields["kind"] = []string{`Must be "ntfy", "gotify" or "webhook".`}
+	}
+
+	if _, err := notify.ParseTarget(body.URL); err != nil {
+		switch {
+		case errors.Is(err, notify.ErrUnsupportedScheme):
+			fields["url"] = []string{"Must start with http:// or https://."}
+		default:
+			fields["url"] = []string{"Must be a full URL including a host, e.g. https://ntfy.example/inventory."}
+		}
+	}
+
+	sendHour := defaultSendHour
+	if body.SendHour != nil {
+		sendHour = *body.SendHour
+	}
+	if sendHour < 0 || sendHour > 23 {
+		fields["send_hour"] = []string{"Must be an hour of the day, 0 to 23."}
+	}
+
+	token, failure := parseNotificationToken(body.Token)
+	if failure != nil {
+		h.errors.WriteError(w, r, failure)
+		return
+	}
+
+	if len(fields) > 0 {
+		h.errors.WriteError(w, r, ValidationFailed(fields, nil))
+		return
+	}
+
+	settings, err := h.store.SaveNotificationSettings(r.Context(), storageID, store.NotificationSettingsInput{
+		Enabled:     body.Enabled,
+		Kind:        body.Kind,
+		URL:         body.URL,
+		Token:       token,
+		SendHour:    sendHour,
+		IncludeSoon: body.IncludeSoon,
+	})
+	if err != nil {
+		h.errors.WriteError(w, r, FromStoreError(err, "storage not found"))
+		return
+	}
+	writeJSON(w, http.StatusOK, newNotificationSettingsResponse(*settings))
+}
+
+// parseNotificationToken maps the request's three shapes onto the store's
+// three states: absent -> nil (keep), null -> "" (clear), string -> replace.
+func parseNotificationToken(raw json.RawMessage) (*string, *Failure) {
+	if raw == nil {
+		return nil, nil
+	}
+	var text *string
+	if err := json.Unmarshal(raw, &text); err != nil {
+		return nil, ValidationFailed(map[string][]string{
+			"token": {"Must be a string, or null to remove the stored token."},
+		}, err)
+	}
+	if text == nil {
+		cleared := ""
+		return &cleared, nil
+	}
+	return text, nil
+}

+// notificationTestResponse is the inline outcome of a test send. A delivery
+// that failed is still a request that succeeded, so it is a 200 carrying
+// ok:false — the settings page shows the reason next to the button rather
+// than an error envelope with nothing specific in it.
+type notificationTestResponse struct {
+	OK     bool   `json:"ok"`
+	Result string `json:"result"`
+}
+
+// Test serves POST /api/storages/{storage_id}/notification-settings/test.
+//
+// It is the difference between finding a typo in a URL now and finding it
+// days later when a digest quietly never arrived, so it reports the outcome
+// inline and writes the same last_result a scheduled run would.
+//
+// It refuses while notifications are disabled. The spec's acceptance criteria
+// are explicit that with enabled = FALSE "no code path can send", and a test
+// button is a code path: enabling is the act that says this endpoint may
+// receive the household's data.
+func (h *NotificationHandler) Test(w http.ResponseWriter, r *http.Request) {
+	storageID, ok := StorageIDFrom(r.Context())
+	if !ok {
+		h.errors.WriteError(w, r, Internal(errNoStorageInContext))
+		return
+	}
+
+	settings, err := h.store.NotificationSettingsFor(r.Context(), storageID)
+	if err != nil {
+		h.errors.WriteError(w, r, FromStoreError(err, "no notification settings for this storage"))
+		return
+	}
+	if !settings.Enabled {
+		h.errors.WriteError(w, r, Conflict("Turn notifications on and save before sending a test.", nil))
+		return
+	}
+
+	if err := h.notifier.SendTest(r.Context(), *settings); err != nil {
+		var delivery *notify.DeliveryError
+		if errors.As(err, &delivery) {
+			writeJSON(w, http.StatusOK, notificationTestResponse{Result: delivery.Summary()})
+			return
+		}
+		h.errors.WriteError(w, r, Internal(err))
+		return
+	}
+	writeJSON(w, http.StatusOK, notificationTestResponse{OK: true, Result: store.NotificationSent})
+}
